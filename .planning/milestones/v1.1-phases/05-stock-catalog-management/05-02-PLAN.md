  <implementation>
    Export async or sync function:
    ```
    export function computeCatalogAggregate(
      catalog: MedicineCatalog,
      activeStocks: Medicine[]
    ): { status: MedicineStatus; totalQty: number } {
      // Find stock with soonest expiryDate (null dates ignored)
      const nearestExpiryStock = activeStocks.reduce(
        (nearest, current) => {
          if (!current.expiryDate) return nearest
          if (!nearest?.expiryDate) return current
          return current.expiryDate < nearest.expiryDate ? current : nearest
        },
        null as Medicine | null
      )

      // Compute status from nearestExpiryStock
      const status = nearestExpiryStock
        ? calculateStatus(nearestExpiryStock)
        : 'Active'

      // Sum quantities (null → 0)
      const totalQty = activeStocks.reduce(
        (sum, stock) => sum + (stock.quantity ?? 0),
        0
      )

      return { status, totalQty }
    }
    ```
