    **1. Add stock entry**
    - Input: catalogId, quantity, quantityUnit, expiryDate, location, medicineName
    - Expected: Row inserted into db.medicines, history entry created, returns new stock ID
    - Edge cases: null expiryDate, null quantity, location null (="Other"), no notes/pao
    - Atomic: both operations succeed or both fail

    **2. Edit stock entry**
    - Input: stockId, changes (Partial<Medicine>), before (Medicine), medicineName
    - Expected: db.medicines row updated, history entry created with changed fields
    - Edge cases: change only quantity, change location, change expiry, null → filled
    - Atomic: both operations succeed or both fail

    **3. Soft-delete stock entry**
    - Input: stockId, Medicine (before), medicineName
    - Expected: deletedAt set to now, history entry created, stock appears in trash
    - Edge cases: stock already deleted (idempotent?), stock with notes, stock with pao

    **4. Move/split stock**
    - Input: stockId (original), quantity to move, targetLocation, medicineName
    - Expected: Original decremented by qty (deletedAt cleared if it was soft-deleted?), new entry created at target location, both atomically
    - Validation: cannot move more than current quantity
    - Edge cases: move all (original qty = 0 after), move 1 from 50, move to same location, move with null quantity on original

    **Edge case assumptions to flag:**
    - Null quantities in sum treated as 0 (flagged)
    - Move quantity validation: no fractional units (whole numbers only)
    - Move to same location: allowed (creates duplicate entry at same location with different ID)
    - Soft-deleted entry: can be moved (move clears deletedAt? or error? — flagged assumption)

    ```typescript
    import { db } from '@/lib/db'
    import type { Medicine, MedicineCatalog } from '@/lib/db'
    import {
      addMedicineHistory,
      updateMedicineWithHistory,
      softDeleteMedicine,
    } from '@/lib/historyOps'

    export async function addStockEntry(
      catalogId: number,
      data: Omit<Medicine, 'id' | 'catalogId' | 'createdAt' | 'updatedAt' | 'deletedAt'>,
      medicineName: string
    ): Promise<number> {
      const now = new Date().toISOString()
      let newId = 0
      await db.transaction('rw', db.medicines, db.history, async () => {
        newId = await db.medicines.add({
          catalogId,
          ...data,
          createdAt: now,
          updatedAt: now,
          deletedAt: null,
        })
        const newMedicine = await db.medicines.get(newId)
        if (newMedicine) {
          await addMedicineHistory(newMedicine, medicineName, 'created')
        }
      })
      return newId
    }

    export async function editStockEntry(
      stockId: number,
      before: Medicine,
      changes: Partial<Medicine>,
      medicineName: string
    ): Promise<void> {
      await updateMedicineWithHistory(
        stockId,
        before,
        { ...changes, updatedAt: new Date().toISOString() },
        medicineName
      )
    }

    export async function softDeleteStock(
      stockId: number,
      stock: Medicine,
      medicineName: string
    ): Promise<void> {
      await softDeleteMedicine(stock, medicineName)
    }

    export async function moveStock(
      stockId: number,
      quantityToMove: number,
      targetLocation: string | null,
      stock: Medicine,
      medicineName: string
    ): Promise<number> {
      if (quantityToMove > (stock.quantity ?? 0)) {
        throw new Error(
          `Cannot move ${quantityToMove} units; only ${stock.quantity} available`
        )
      }

      const now = new Date().toISOString()
      let newId = 0
      await db.transaction('rw', db.medicines, db.history, async () => {
        // Decrement original
        const newOriginalQty = (stock.quantity ?? 0) - quantityToMove
        await db.medicines.update(stockId, {
          quantity: newOriginalQty,
          updatedAt: now,
        })

        // Create new entry at target location
        newId = await db.medicines.add({
          catalogId: stock.catalogId,
          quantity: quantityToMove,
          quantityUnit: stock.quantityUnit,
          expiryDate: stock.expiryDate,
          openedDate: stock.openedDate,
          pao: stock.pao,
          location: targetLocation,
          manualStatus: null,
          notes: stock.notes,
          createdAt: now,
          updatedAt: now,
          deletedAt: null,
        })

        // Record history for both
        const updatedStock = await db.medicines.get(stockId)
        if (updatedStock) {
          await updateMedicineWithHistory(
            stockId,
            stock,
            { quantity: newOriginalQty, updatedAt: now },
            medicineName
          )
        }

        const newStock = await db.medicines.get(newId)
        if (newStock) {
          await addMedicineHistory(newStock, medicineName, 'created')
        }
      })
      return newId
    }
    ```
