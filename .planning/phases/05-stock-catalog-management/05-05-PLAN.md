    export function MedicineNew() {
      const navigate = useNavigate()
      
      // Step state machine
      const [step, setStep] = useState<'search' | 'create-catalog' | 'stock-form'>('search')
      const [selectedCatalog, setSelectedCatalog] = useState<MedicineCatalog | null>(null)
      const [searchText, setSearchText] = useState('')
      
      // Handler: select existing catalog
      function handleCatalogSelect(catalog: MedicineCatalog) {
        setSelectedCatalog(catalog)
        setStep('stock-form')
      }
      
      // Handler: user clicked "Create '[name]'" option
      function handleCreateCatalogClick(typedName: string) {
        setSearchText(typedName)
        setStep('create-catalog')
      }
      
      // Handler: catalog created in Step 2
      async function handleCatalogCreated(newCatalog: MedicineCatalog) {
        setSelectedCatalog(newCatalog)
        setStep('stock-form')
      }
      
      // Handler: back arrow in Step 2/3 (return to search, discard data)
      function handleBackToSearch() {
        setStep('search')
        setSearchText('')
      }
      
      // Handler: stock form submitted
      async function handleStockSubmit(stockData: Partial<Medicine>) {
        if (!selectedCatalog) return
        
        try {
          const stockId = await addStockEntry(
            selectedCatalog.id,
            {
              quantity: stockData.quantity ?? null,
              quantityUnit: stockData.quantityUnit ?? null,
              expiryDate: stockData.expiryDate,
              location: stockData.location ?? null,
              openedDate: stockData.openedDate ?? null,
              pao: stockData.pao ?? null,
              manualStatus: null,
              notes: stockData.notes ?? null,
            },
            selectedCatalog.name
          )
          
          // Navigate to detail view
          void navigate(`/medicines/${selectedCatalog.id}`)
        } catch (err) {
          console.error('Failed to add stock:', err)
          toast.error('Failed to save. Please try again.')
        }
      }
      
      // Handler: cancel (any step)
      function handleCancel() {
        void navigate('/medicines')
      }
      
      return (
        <div>
          {step === 'search' && (
            <CatalogAutocomplete
              onSelect={handleCatalogSelect}
              onCreateClick={handleCreateCatalogClick}
            />
          )}
          
          {step === 'create-catalog' && (
            <div className="p-4 space-y-4">
              <div className="flex items-center gap-2">
                <button onClick={handleBackToSearch}>← Back</button>
                <h2 className="text-lg font-semibold">Create Medicine</h2>
              </div>
              
              <form
                onSubmit={(e) => {
                  e.preventDefault()
                  // Create catalog and call handleCatalogCreated
                  // Use CatalogFields here or embed form
                }}
              >
                {/* CatalogFields component */}
                {/* OR inline form with name + category fields */}
                <div className="space-y-4">
                  <Input placeholder="Medicine name" defaultValue={searchText} />
                  <Select>
                    <option>Select category...</option>
                    {/* Categories */}
                  </Select>
                  <Input placeholder="Form (optional)" />
                  <Textarea placeholder="Notes (optional)" />
                  <div className="flex gap-2">
                    <Button type="button" variant="outline" onClick={handleBackToSearch}>
                      Cancel
                    </Button>
                    <Button type="submit" className="flex-1">Create catalog</Button>
                  </div>
                </div>
              </form>
            </div>
          )}
          
          {step === 'stock-form' && selectedCatalog && (
            <div className="p-4 space-y-4">
              <div className="flex items-center gap-2">
                <button onClick={handleBackToSearch}>← Back</button>
                <h2 className="text-lg font-semibold">Add Stock: {selectedCatalog.name}</h2>
              </div>
              
              <form
                onSubmit={(e) => {
                  e.preventDefault()
                  // handleStockSubmit from form data
                }}
              >
                {/* StockFields component */}

                  {/* Location select */}
                  {/* PAO, notes, etc. */}

                      Cancel
                    </Button>
                    <Button type="submit" className="flex-1">Add stock</Button>
                  </div>
                </div>
              </form>
            </div>
          )}
        </div>
      )
    }
